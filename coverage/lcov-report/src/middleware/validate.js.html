function validateIdParam(req, res, next) {
    const id = Number(req.params.id);
    if (!Number.isInteger(id) || id <= 0) {
        return res.status(400).json({ error: 'Некоректний числовий ідентифікатор ресурсу.' });
    }
    req.resourceId = id;
    next();
}
